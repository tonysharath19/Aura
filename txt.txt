    function initOrb3D() {
      const canvas = document.getElementById('orbCanvas');
      const ctx = canvas.getContext('2d');
      let w, h;

      function resize() {
        w = canvas.width = canvas.parentElement.clientWidth;
        h = canvas.height = canvas.parentElement.clientHeight;
      }
      resize();
      window.addEventListener('resize', resize);

      const numParticles = 600;
      const particles = [];
      const radius = 115;

      for (let i = 0; i < numParticles; i++) {
        const theta = Math.random() * Math.PI * 2;
        const phi = Math.acos((Math.random() * 2) - 1);
        particles.push({
          x: radius * Math.sin(phi) * Math.cos(theta),
          y: radius * Math.sin(phi) * Math.sin(theta),
          z: radius * Math.cos(phi),
          color: Math.random() > 0.4 ? '#00f2fe' : (Math.random() > 0.5 ? '#7f00ff' : '#3b82f6')
        });
      }

      let rotX = 0.002;
      let rotY = 0.004;
      let mouseX = 0, mouseY = 0;

      document.addEventListener('mousemove', (e) => {
        mouseX = (e.clientX - window.innerWidth / 2) * 0.00004;
        mouseY = (e.clientY - window.innerHeight / 2) * 0.00004;
      });

      function render() {
        ctx.clearRect(0, 0, w, h);
        const cx = w / 2;
        const cy = h / 2;

        // Core Glowing Aura
        const auraGrad = ctx.createRadialGradient(cx, cy, 10, cx, cy, radius * 1.35);
        auraGrad.addColorStop(0, 'rgba(0, 242, 254, 0.4)');
        auraGrad.addColorStop(0.4, 'rgba(59, 130, 246, 0.2)');
        auraGrad.addColorStop(0.8, 'rgba(127, 0, 255, 0.1)');
        auraGrad.addColorStop(1, 'rgba(0, 0, 0, 0)');
        
        ctx.fillStyle = auraGrad;
        ctx.beginPath();
        ctx.arc(cx, cy, radius * 1.4, 0, Math.PI * 2);
        ctx.fill();

        // Rotate & Render Particle Sphere
        const cX = rotX + mouseY;
        const cY = rotY + mouseX;

        particles.forEach(p => {
          let x1 = p.x * Math.cos(cY) - p.z * Math.sin(cY);
          let z1 = p.z * Math.cos(cY) + p.x * Math.sin(cY);
          let y1 = p.y * Math.cos(cX) - z1 * Math.sin(cX);
          let z2 = z1 * Math.cos(cX) + p.y * Math.sin(cX);

          p.x = x1; p.y = y1; p.z = z2;

          const scale = 320 / (320 + p.z);
          const px = p.x * scale + cx;
          const py = p.y * scale + cy;
          const pSize = Math.max(0.6, (p.z + radius) / (radius * 2) * 2.2);
          const alpha = Math.max(0.15, (p.z + radius) / (radius * 2));

          ctx.fillStyle = p.color;
          ctx.globalAlpha = alpha;
          ctx.beginPath();
          ctx.arc(px, py, pSize, 0, Math.PI * 2);
          ctx.fill();
        });

        // Outer Elliptical Orbital Rings
        ctx.globalAlpha = 0.5;
        ctx.strokeStyle = '#00f2fe';
        ctx.lineWidth = 1.2;
        ctx.beginPath();
        ctx.ellipse(cx, cy, radius * 1.3, radius * 0.4, Math.PI / 5, 0, Math.PI * 2);
        ctx.stroke();

        ctx.globalAlpha = 0.3;
        ctx.strokeStyle = '#7f00ff';
        ctx.beginPath();
        ctx.ellipse(cx, cy, radius * 1.45, radius * 0.45, -Math.PI / 4, 0, Math.PI * 2);
        ctx.stroke();

        requestAnimationFrame(render);
      }
      render();
    }

    function initWaveMesh() {
      const canvas = document.getElementById('waveCanvas');
      const ctx = canvas.getContext('2d');
      let w, h;

      function resize() {
        w = canvas.width = window.innerWidth;
        h = canvas.height = 180;
      }
      resize();
      window.addEventListener('resize', resize);

      let offset = 0;
      function drawWave() {
        ctx.clearRect(0, 0, w, h);
        offset += 0.012;

        ctx.beginPath();
        ctx.lineWidth = 1.5;
        const grad = ctx.createLinearGradient(0, 0, w, 0);
        grad.addColorStop(0, 'rgba(0, 242, 254, 0)');
        grad.addColorStop(0.5, 'rgba(127, 0, 255, 0.45)');
        grad.addColorStop(1, 'rgba(0, 242, 254, 0)');
        ctx.strokeStyle = grad;

        for (let x = 0; x < w; x += 15) {
          const y = Math.sin(x * 0.006 + offset) * 20 + Math.cos(x * 0.004 + offset * 0.7) * 12 + (h / 1.5);
          if (x === 0) ctx.moveTo(x, y);
          else ctx.lineTo(x, y);
        }
        ctx.stroke();

        requestAnimationFrame(drawWave);
      }
      drawWave();
    }

    function renderSparkline(canvasId, points, strokeColor, fillColor) {
      const canvas = document.getElementById(canvasId);
      if (!canvas) return;
      const ctx = canvas.getContext('2d');
      const w = canvas.width = canvas.parentElement.clientWidth;
      const h = canvas.height = 36;

      ctx.clearRect(0, 0, w, h);

      const min = Math.min(...points);
      const max = Math.max(...points);
      const range = (max - min) || 1;

      // Fill Path
      ctx.beginPath();
      points.forEach((p, i) => {
        const x = (i / (points.length - 1)) * w;
        const y = h - ((p - min) / range) * (h - 8) - 4;
        if (i === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
      });
      
      ctx.strokeStyle = strokeColor;
      ctx.lineWidth = 2;
      ctx.stroke();

      if (fillColor) {
        ctx.lineTo(w, h);
        ctx.lineTo(0, h);
        ctx.closePath();
        ctx.fillStyle = fillColor;
        ctx.fill();
      }
    }

    function renderBarChart(canvasId, values, barColor) {
      const canvas = document.getElementById(canvasId);
      if (!canvas) return;
      const ctx = canvas.getContext('2d');
      const w = canvas.width = canvas.parentElement.clientWidth;
      const h = canvas.height = 36;

      ctx.clearRect(0, 0, w, h);
      const barWidth = 4;
      const gap = (w - (values.length * barWidth)) / (values.length - 1);

      const max = Math.max(...values);

      values.forEach((val, i) => {
        const barHeight = (val / max) * (h - 6);
        const x = i * (barWidth + gap);
        const y = h - barHeight;

        ctx.fillStyle = barColor;
        ctx.beginPath();
        ctx.roundRect(x, y, barWidth, barHeight, 2);
        ctx.fill();
      });
    }
